<Snippet id="numpy-init">

```python
import numpy as np

rng = np.random.default_rng(42)
w = rng.normal(0.0, 1.0 / np.sqrt(2), size=2)
b = 0.0
```

</Snippet>

<Snippet id="pytorch-forward">


</Snippet>

<Snippet id="numpy-forward">

```python
def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-z))

p = sigmoid(X @ w + b)
```

</Snippet>

{/* ---- Loss ---- */}

<Snippet id="sklearn-loss">

```python

```

</Snippet>

<Snippet id="pytorch-loss">

```python
loss_fn = nn.BCEWithLogitsLoss()

loss = loss_fn(z, y)  # hand it the logit, so no sigmoid above
```

</Snippet>

<Snippet id="numpy-loss">

```python
eps = 1e-7
p = np.clip(p, eps, 1.0 - eps)
loss = -np.mean(y * np.log(p) + (1 - y) * np.log(1 - p))
```

</Snippet>

{/* ---- Optimiser ---- */}

<Snippet id="sklearn-optimiser">

```python

```

</Snippet>

<Snippet id="pytorch-optimiser">

```python
opt = torch.optim.SGD(layer.parameters(), lr=0.5)

loss.backward()  # fills in every .grad
opt.step()       # w -= lr * w.grad
opt.zero_grad()
```

</Snippet>

<Snippet id="numpy-optimiser">

```python
error = p - y                      # the cancellation, again
grad_w = X.T @ error / len(y)
grad_b = error.mean()

w -= lr * grad_w
b -= lr * grad_b
```

</Snippet>
